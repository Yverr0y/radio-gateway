# ============================================================================
# RADIO GATEWAY - CONFIGURATION FILE (INI format)
# ============================================================================
#
# Bidirectional audio gateway between Mumble VoIP and radio via AIOC
#
# ─── Fresh install? Minimum to get running ─────────────────────────────────
#
# REQUIRED (gateway won't connect without these):
#   [mumble] MUMBLE_SERVER         — hostname/IP of your Mumble server
#   [mumble] MUMBLE_PORT           — default 64738
#   [mumble] MUMBLE_USERNAME       — display name on Mumble
#   [mumble] MUMBLE_PASSWORD       — leave blank if server has no password
#
# RECOMMENDED (enables key optional features):
#   [stream_output] STREAM_*       — Broadcastify feed key (sign up at
#                                    broadcastify.com)
#   [telegram] TELEGRAM_BOT_TOKEN  — create a bot via @BotFather on Telegram
#   [telegram] TELEGRAM_CHAT_ID    — get via the "getUpdates" curl once
#                                    you've messaged your bot
#   [gdrive] GDRIVE_*              — OAuth in the web UI at /gdrive
#
# RADIO HARDWARE (enable only what you own):
#   [radio] / [th9800_cat]         — TH-9800 via AIOC USB
#   [kv4p] / [d75]                 — KV4P HT or Kenwood D75
#   [sdr_plugin]                   — RSPduo or RTL-SDR via rtl_airband
#
# See INSTALL.md for a full walk-through, per-credential instructions, and
# troubleshooting.
#
# Quick Start:
#   1. Set MUMBLE_SERVER to your server IP/hostname
#   2. Most other settings have sensible defaults
#   3. Optional features are disabled by default — enable as needed
#
# Keyboard Controls (runtime):
#   MUTE:  't'=TX  'r'=RX  'm'=Global  's'=SDR1  'x'=SDR2  'c'=Remote  'a'=Announce  'o'=Speaker
#   AUDIO: 'v'=VAD toggle  ','=Vol-  '.'=Vol+
#   PROC:  'n'=Gate  'f'=HPF  'g'=AGC
#   SDR:   'd'=SDR1 Duck toggle
#   PTT:   'p'=Manual PTT toggle
#   PLAY:  '1-9'=Announcements  '0'=Station ID  '-'=Stop
#   NET:   'k'=Reset remote audio connection
#   RELAY: 'j'=Radio power button  'h'=Charger toggle
#   SMART: '['=Smart#1  ']'=Smart#2  '\'=Smart#3
#   TRACE: 'i'=Start/stop audio trace  'u'=Start/stop watchdog trace
#   MISC:  'q'=Restart gateway  'z'=Clear and reprint console
# ============================================================================

[ptt]
# Push-to-talk control
#
# Three PTT methods available:
#   aioc     = AIOC USB HID GPIO (default, requires AIOC device)
#   relay    = CH340 USB relay module (separate from radio power/charger relays)
#   software = CAT TCP command to TH9800_CAT.py (!rts)
#
# Status bar shows PTT method: PTT: (AIOC), PTTR: (relay), PTTS: (software)

PTT_METHOD = aioc

# PTT Relay settings (only used when PTT_METHOD = relay)
PTT_RELAY_DEVICE = /dev/relay_ptt
PTT_RELAY_BAUD = 9600

PTT_ACTIVATION_DELAY = 0.1   # Seconds before keying PTT after audio detected
PTT_RELEASE_DELAY = 0.5      # Seconds to hold PTT after audio stops

# TX Talkback — route TX audio (TTS/CW/announcements) to local outputs
# When False (default), TX audio goes only to the radio — local listeners
# (speaker, Mumble, PCM/MP3 stream) hear silence or concurrent RX instead.
# When True, TX audio is also played through local outputs for monitoring.
TX_TALKBACK = False

# Delay after PTT key-up before announcement/TTS audio starts (seconds)
# Gives the radio time to fully key up before speech begins
PTT_ANNOUNCEMENT_DELAY = 0.5

# Auto-PTT for link endpoints wired as a bus TX sink.
# THRESHOLD is the bus output level (0-100) that keys the endpoint's radio:
#   raise it if a noisy source keys unintentionally, lower it if a quiet
#   source never keys at all.
# HOLD is how long the key is held after the last chunk above the threshold:
#   raise it to stop the radio re-keying during gaps in speech.
LINK_AUTO_PTT_THRESHOLD = 10
LINK_AUTO_PTT_HOLD = 0.5

# Pre-key TX buffer for bus PTT (milliseconds, 0 = disabled).
# PTT is applied off-tick and a radio takes 150-600 ms to key, so TX audio
# produced during that window would be pushed into a radio that isn't
# transmitting yet — the first syllable of every transmission. Audio waits
# in this buffer instead and is flushed, in order, once the key lands.
# Bounded: oldest audio drops past this depth, so a slow or failed key can't
# build unbounded latency.
PTT_PREKEY_BUFFER_MS = 500

# Jitter buffer depth, in 50 ms chunks, before a remote source starts playing.
# This is the single biggest contributor to end-to-end latency, so it is worth
# tuning — but tune it with evidence, not by ear:
#   LINK_JITTER_PREFILL   4 = 200 ms cushion. A wired-LAN endpoint usually runs
#                         fine at 2 (100 ms). Keep 4+ for WiFi or a
#                         Cloudflare-tunnelled endpoint.
#   REMOTE_AUDIO_JITTER_PREFILL  8 = 400 ms. Higher because the Windows client's
#                         WASAPI callback gaps are burstier than a link endpoint.
# After lowering either, watch rg_link_audio_underruns_total in Prometheus: a
# rising count is the dropout cost appearing before it becomes audible.
# Individual link endpoints can override LINK_JITTER_PREFILL at runtime (the
# value is remembered per endpoint) — this is only the default for new ones.
LINK_JITTER_PREFILL = 4
REMOTE_AUDIO_JITTER_PREFILL = 8

[streaming]
# Broadcastify / Icecast streaming
#
# Requires: ffmpeg (installed by scripts/install.sh).
# Audio flow: Gateway -> in-process ffmpeg MP3 encoder -> Icecast HTTP SOURCE
#
# Since v4.1.0 this runs entirely inside the gateway process. There is no
# ALSA loopback and no darkice process. The legacy external-darkice path is
# still available behind SUPERVISE_DARKICE (see [advanced]) but is off by
# default and reads /etc/darkice.cfg, not the STREAM_* keys below.

ENABLE_STREAM_OUTPUT = false
STREAM_SERVER = audio9.broadcastify.com
STREAM_PORT = 80
STREAM_PASSWORD = your_feed_password
STREAM_MOUNT = /your_feed_mount
STREAM_NAME = Radio Gateway
STREAM_DESCRIPTION = Radio to Mumble Gateway Stream
STREAM_FORMAT = mp3       # mp3 or ogg
# Broadcastify: 16 kbps per scanner. A dual-channel feed is 32 kbps TOTAL,
# i.e. the same 16k per scanner — not better quality per channel.
# Set 32 when STREAM_DUAL_CHANNEL is true, 16 when it is false.
STREAM_BITRATE = 32       # kbps (16 = single scanner, 32 = dual-channel feed)

# Encoder output sample rate. MUST stay below 32000. At 32/44.1/48 kHz the
# encoder is MPEG-1 Layer III, whose lowest Layer III bitrate is 32 kbps, so
# lame silently clamps anything smaller and STREAM_BITRATE is ignored. 16 kbps
# only exists in MPEG-2 Layer III (16/22.05/24 kHz). 22.05 kHz carries 11 kHz
# of audio; narrowband FM voice is ~3 kHz, so nothing audible is lost.
STREAM_SAMPLE_RATE = 22050

# Dual-channel (two-scanner) Broadcastify feed.
#   true  -> "Broadcastify [L]" and "Broadcastify [R]" sinks on the routing
#            page. Route a different bus to each; an unrouted side is filled
#            with silence. Needs STREAM_BITRATE = 32.
#   false -> a single mono "Broadcastify" sink. Needs STREAM_BITRATE = 16.
# The two node sets are mutually exclusive by construction, so the routing
# graph cannot express an invalid combination. Changing this orphans any
# existing Broadcastify connection in routing_config.json — rewire it on the
# /routing page afterwards, or the feed goes silent.
# Broadcastify must also have the feed provisioned for two sources at their
# end; the encoder change alone will not make them split it.
STREAM_DUAL_CHANNEL = true

# Seconds to wait before reconnecting when Icecast is still holding our own
# previous source connection (a detected drop, or a 403 "mount in use"). The
# server has to reap the stale source before it will accept a new one;
# measured clear time was under 10s, so 15 gives headroom. Ordinary connect
# failures use the normal backoff instead of this longer wait.
STREAM_MOUNT_WAIT = 15

# A pinned worker is never TTL-expired: it stays in the pool showing
# 'unreachable' when down, because the operator asked for it explicitly.

# Worker self-registration (v4.2.0). Workers started with --gateway dial into

# ============================================================================
# TROUBLESHOOTING
# ============================================================================
#
# AUDIO:
#   No audio on Mumble     -> Check INPUT_VOLUME, VAD/VOX settings
#   No audio on radio      -> Check OUTPUT_VOLUME, AIOC connection
#   Choppy / -9999 errors  -> Increase AUDIO_CHUNK_SIZE
#
# MUMBLE:
#   Won't connect          -> Verify MUMBLE_SERVER and MUMBLE_PORT
#   Channel not found      -> Check MUMBLE_CHANNEL spelling
#
# PTT:
#   Doesn't activate       -> Try AIOC_PTT_CHANNEL = 1, 2, or 3
#   Cuts off too fast      -> Increase PTT_RELEASE_DELAY
#
# SDR:
#   Goes silent after hours -> Watchdog will auto-recover (see SDR_WATCHDOG_*)
#   Status shows W1, W2    -> Recovery attempts made; !restart if still silent
#
# STREAMING:
#   Stream offline          -> Check the Broadcastify card on /dashboard/services
#                              for the last error; the encoder is in-process, so
#                              there is no darkice process to look for
#   No audio on stream      -> Check the Broadcastify sink is routed on /routing
#   Bitrate stuck at 32k    -> STREAM_SAMPLE_RATE must stay below 32000
#
# For more help, enable VERBOSE_LOGGING = true
# ============================================================================
